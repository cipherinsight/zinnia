@zk_circuit
def crc16(data: Integer[Array, '16'], poly: Integer=33800):
    """
    CRC-16-CCITT Algorithm
    """
    crc = 65535
    for b in data:
        cur_byte = 255 & b
        for _ in range(0, 8):
            if crc & 1 ^ cur_byte & 1:
                crc = crc >> 1 ^ poly
            else:
                crc >>= 1
            cur_byte >>= 1
    crc = ~crc & 65535
    crc = crc << 8 | crc >> 8 & 255
    return crc & 65535

@zk_circuit
def CROTG(CA: Complex, CB: Complex, C: Float=0, S: Complex=0):
    if abs(CA) == 0.0:
        C = 0.0
        S = complex(1.0, 0.0)
        CA = CB
    else:
        SCALE = abs(CA) + abs(CB)
        NORM = SCALE * math.sqrt(abs(CA / SCALE) ** 2 + abs(CB / SCALE) ** 2)
        ALPHA = CA / abs(CA)
        C = abs(CA) / NORM
        S = ALPHA * CB.conjugate() / NORM
        CA = ALPHA * NORM
    return (CA, CB, C, S)

        return v + n
    elif v < n:
        return v
    else:
        return v - n

@zk_chip
def _ising_update(field, n, m, beta):
    total = 0
    N, M = field.shape
    for i in range(n - 1, n + 2):
        for j in range(m - 1, m + 2):
            if i == n and j == m:
                continue
            total += field[clamp(i, N), clamp(j, M)]
    dE = 2 * field[n, m] * total
    if dE <= 0:
        field[n, m] *= -1
    elif np.exp(-dE * beta) > 0.5:
        field[n, m] *= -1

    periodic_laplacian_1D_4th_order
    Implementing the 4th order 1D laplacian with periodic condition
    """
    lap = numpy.zeros_like(var)
    lap[1:] = 4.0 / 3.0 * var[:-1]
    lap[0] = 4.0 / 3.0 * var[1]
    lap[:-1] += 4.0 / 3.0 * var[1:]

    lap[-2:] += -1.0 / 12.0 * var[:2]

### pythran__pythran_periodic_dist__dist
# pythran program periodic_dist.py (serge-sans-paille/pythran pythran/tests/cases).
# Function bodies are the upstream text. Entry-function annotations are derived from upstream
# types by benchmarking/datasets/tools/acceptance_case_set.py; see datasets/case_sets/acceptance.json.

from zinnia import *
import numpy as np

@zk_circuit
def dist(x: Float[Array, '16'], y: Float[Array, '16'], z: Float[Array, '16'], L: Integer, periodicX: Boolean, periodicY: Boolean, periodicZ: Boolean):
    """ Computes distances between all particles and places the result in a matrix such that the ij th matrix entry corresponds to the distance between particle i and j """
    N = len(x)
    xtemp = np.tile(x, (N, 1))
    dx = xtemp - xtemp.T
    ytemp = np.tile(y, (N, 1))
    dy = ytemp - ytemp.T
    ztemp = np.tile(z, (N, 1))
    dz = ztemp - ztemp.T
    if periodicX:
        dx[dx > L / 2] = dx[dx > L / 2] - L
        dx[dx < -L / 2] = dx[dx < -L / 2] + L
    if periodicY:
        dy[dy > L / 2] = dy[dy > L / 2] - L
        dy[dy < -L / 2] = dy[dy < -L / 2] + L
    if periodicZ:
        dz[dz > L / 2] = dz[dz > L / 2] - L
        dz[dz < -L / 2] = dz[dz < -L / 2] + L
    d = np.sqrt(dx ** 2 + dy ** 2 + dz ** 2)
    d[d == 0] = -1
    return (d, dx, dy, dz)

@zk_circuit
def permutations(iterable: Integer[Array, '4']):
    """permutations(range(3), 2) --> (0,1) (0,2) (1,0) (1,2) (2,0) (2,1)"""
    out = []
    pool = tuple(iterable)
    n = len(pool)
    r = n
    indices = list(range(n))
    cycles = list(range(n - r + 1, n + 1))[::-1]
    out.append(tuple([pool[i] for i in indices[:r]]))
    while 1:
        for i in reversed(range(r)):
            cycles[i] -= 1
            if cycles[i] == 0:
                indices[i:] = indices[i + 1:] + indices[i:i + 1]
                cycles[i] = n - i
            else:
                j = cycles[i]
                indices[i], indices[-j] = (indices[-j], indices[i])
                out.append(tuple([pool[i] for i in indices[:r]]))
                break
        else:
            return out

        pzi = points[i, 2]
        total = 0.0
        for j in range(n_weights):
            weight_j = weights[j]
            xj = pos[j, 0]
            yj = pos[j, 1]
            zj = pos[j, 2]
            dx = pxi - pos[j, 0]
            dy = pyi - pos[j, 1]
            dz = pzi - pos[j, 2]
            dr = 1.0 / np.sqrt(dx * dx + dy * dy + dz * dz)
            total += weight_j * dr
            sum_array3d[i, 0] += weight_j * dx
            sum_array3d[i, 1] += weight_j * dy
            sum_array3d[i, 2] += weight_j * dz
        return total
    sum_array = np.array([compute(i) for i in range(n_points)])
    return (sum_array, sum_array3d)

@zk_circuit
def slow_wdist(A: Float[Array, '2 3'], B: Float[Array, '2 3'], W: Float[Array, '2 3']):
    k, m = A.shape
    _, n = B.shape
    D = np.zeros((m, n))

### pythran__pythran_weights__weights
# pythran program weights.py (serge-sans-paille/pythran pythran/tests/cases).
# Function bodies are the upstream text. Entry-function annotations are derived from upstream
# types by benchmarking/datasets/tools/acceptance_case_set.py; see datasets/case_sets/acceptance.json.

from zinnia import *
import numpy as np

@zk_circuit
def weights(input_data: Integer[Array, '5 2'], threshold: Float=0.3):
    n_seq, length = input_data.shape
    weights = np.zeros(n_seq, dtype=np.float32)
    for i in range(n_seq):
        vector = input_data[i, None, :]
        count_matches = np.sum(vector == input_data, axis=1)
        over_threshold = count_matches > threshold * length
        total = np.sum(over_threshold)
        weights[i] = 1 / total
    return weights
